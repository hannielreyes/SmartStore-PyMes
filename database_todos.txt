def _normalizar_texto(valor):
    if valor is None:
        return ""
    return str(valor).strip()


def _show_full_image(page: ft.Page, src: str, src_2: str | None = None):
    """PANTALLA AUXILIAR: muestra im�genes del producto con zoom."""
    imagenes = [imagen for imagen in (src, src_2) if imagen]
    indice = 0
    previous_controls = list(page.controls)
    imagen_control = ft.Image(src=imagenes[0], fit="contain", expand=True)

    def siguiente(e):
        nonlocal indice
        indice = (indice + 1) % len(imagenes)
        imagen_control.src = imagenes[indice]
        page.update()

    def cerrar(e):
        page.clean()
        page.bgcolor = "white"
        for control in previous_controls:
            page.add(control)
        page.update()

    ancho_disponible = max((page.width or 360) - 32, 260)
    ancho_imagen = min(ancho_disponible, 600)
    ancho_botones = min(ancho_disponible, 420)

    # BOTONES: permite ver la siguiente imagen o cerrar el visor.
    botones = ft.Column(
        [
            ft.ElevatedButton(
                "Ver siguiente imagen ?",
                on_click=siguiente,
                visible=len(imagenes) > 1,
                bgcolor="#040D4A",
                color="white",
                height=44,
                width=ancho_botones,
            ),
            ft.ElevatedButton(
                "Cerrar",
                on_click=cerrar,
                bgcolor="#EFA520",
                color="#040D4A",
                height=44,
                width=ancho_botones
            ),
        ],
        spacing=12,
        width=ancho_botones,
        alignment=ft.MainAxisAlignment.CENTER,
        horizontal_alignment=ft.CrossAxisAlignment.CENTER,
    )

    vista_imagen = ft.Container(
        expand=True,
        image=ft.DecorationImage(src="fondopro.png", fit="cover"),
        padding=20,
        content=ft.Column(
            [
                ft.Container(
                    content=ft.InteractiveViewer(
                        content=imagen_control,
                        min_scale=1,
                        max_scale=4,
                        scale_enabled=True,
                        pan_enabled=True,
                    ),
                    width=ancho_imagen,
                    height=560,
                    border_radius=24,
                    bgcolor="transparent",
                    padding=0,
                ),
                ft.Container(
                    content=botones,
                    padding=12,
                    border_radius=24,
                    bgcolor="rgba(255,255,255,0.18)",
                    alignment=ft.Alignment(0, 0),
                    width=min(ancho_disponible, 420),
                ),
            ],
            alignment=ft.MainAxisAlignment.CENTER,
            horizontal_alignment=ft.CrossAxisAlignment.CENTER,
            expand=True,
            scroll=ft.Scrollbar(thickness=4, radius=4, interactive=True),
        ),
    )

    page.clean()
    page.bgcolor = "white"
    page.add(vista_imagen)
    page.update()

def carrito_view(page: ft.Page, usuario, go_back):
    # PANTALLA: reservas del cliente, pagos, estados y acciones de cada pedido.
    configurar_retroceso(page, lambda: go_back(usuario))
    page.clean()
    page.bgcolor = "white"
    db.cancelar_reservas_vencidas()

    fondo = ft.Container(
        expand=True,
        image=ft.DecorationImage(src="fondopro.png", fit="cover"),
    )

    # CONTROLES: espacio desplazable donde se apilan las tarjetas de reservas.
    # El tamaño de cada tarjeta se mantiene independiente de este contenedor.
    lista = ft.Column(
        spacing=10,
        scroll=ft.Scrollbar(thickness=4, radius=4, interactive=True),
        height=500,
    )
    total_text = ft.Text("Total: C$0.00 / $0.00", weight=ft.FontWeight.BOLD)
    verificaciones_en_curso = set()
    verificaciones_lock = threading.Lock()
    contexto = getattr(page, "_smartstore_context", None) or {}
    reservas_precargadas = contexto.get("reservas_usuario")
    usar_reservas_precargadas = reservas_precargadas is not None

    def _ejecutar_captura_paypal(e, reserva_id, tienda_id, total):
        # ACCIÓN: verifica y captura en PayPal el pago de una reserva.
        print(f"🔄 Intentando verificar pago para reserva #{reserva_id}, tienda #{tienda_id}, total={total}...")
        order_id = db.obtener_order_id(reserva_id)
        print(f"   order_id obtenido: {order_id}")

        if not order_id:
            mostrar_snackbar(
                page,
                "⚠️ No se encontró una orden de PayPal. Toca 'Pagar con PayPal' primero.",
                color="orange",
            )
            return

        try:
            mostrar_snackbar(page, "⏳ Verificando pago con PayPal...", color="blue")

            estado_paypal = getattr(
                client.execute(OrdersGetRequest(order_id)).result,
                "status",
                None,
            )
            if estado_paypal != "APPROVED":
                mostrar_snackbar(
                    page,
                    f"⚠️ PayPal aún no aprobó la orden ({estado_paypal}). Abre 'Pagar con PayPal', completa el pago y vuelve a verificar.",
                    color="orange",
                    duracion=7000,
                )
                return

            request = OrdersCaptureRequest(order_id)
            print(f"   Enviando captura a PayPal para orden: {order_id}")
            response = client.execute(request)
            print(f"   Respuesta PayPal: status_code={response.status_code}")

            if response.status_code in (200, 201):
                print("✅ Captura exitosa! Procesando pago en BD...")
                exito, mensaje = db.procesar_pago_reserva(reserva_id, total, tienda_id)
                print(f"   Resultado BD: exito={exito}, mensaje={mensaje}")
                if exito:
                    mostrar_snackbar(
                        page,
                        "✅ Pago verificado correctamente.",
                        color="green",
                    )
                    cargar()
                else:
                    mostrar_snackbar(page, f"⚠️ {mensaje}", color="orange")
        except Exception as err:
            print(f"❌ Error en ejecutar_captura_paypal: {err}")
            if "RESOURCE_NOT_FOUND" in str(err) or "INVALID_RESOURCE_ID" in str(err):
                db.guardar_order_id(reserva_id, None)
                mostrar_snackbar(
                    page,
                    "⚠️ La orden de PayPal ya no existe. Toca 'Pagar con PayPal' para crear una nueva.",
                    color="orange",
                )
                return
            if "ORDER_NOT_APPROVED" in str(err):
                mostrar_snackbar(
                    page,
                    "⚠️ El pago aún no fue aprobado en PayPal. Completa el pago y vuelve a verificar.",
                    color="orange",
                    duracion=7000,
                )
                return
            import traceback
            traceback.print_exc()
            mostrar_snackbar(page, ft.Text(f"❌ Error al verificar: {str(err)}", color="red"))

    def ejecutar_captura_paypal(e, reserva_id, tienda_id, total):
        with verificaciones_lock:
            if reserva_id in verificaciones_en_curso:
                mostrar_snackbar(page, "⏳ Ya se está verificando esta reserva.", color="blue")
                return
            verificaciones_en_curso.add(reserva_id)
        try:
            _ejecutar_captura_paypal(e, reserva_id, tienda_id, total)
        finally:
            with verificaciones_lock:
                verificaciones_en_curso.discard(reserva_id)

    def procesar_pago_simulado(reserva_id, tienda_id, total, dlg=None):
        # ACCIÓN: procesa un pago simulado para pruebas o entornos sin PayPal.
        if dlg is not None:
            dlg.open = False
            page.update()

        try:
            exito, mensaje = db.procesar_pago_reserva(reserva_id, total, tienda_id)
            if exito:
                mostrar_snackbar(
                    page,
                    ft.Text(
                        f"✅ Pago simulado exitoso. Se agregó C${total:.2f} a la billetera.",
                        color="green",
                    ),
                    duration=4000,
                )
                cargar()
            else:
                mostrar_snackbar(page, f"⚠️ {mensaje}", color="orange")
        except Exception as e:
            mostrar_snackbar(page, f"❌ Error simulado: {str(e)}", color="red")

    def procesar_pago_real(reserva_id, tienda_id, total, paypal_email, dlg=None):
        # ACCIÓN: crea o reutiliza una orden real de PayPal.
        if dlg is not None:
            dlg.open = False
            page.update()

        try:
            estado_reserva = db.obtener_estado_reserva(reserva_id)
            if estado_reserva != "Reservado":
                mostrar_snackbar(
                    page,
                    f"✅ Esta reserva ya está en estado: {estado_reserva or 'procesado'}. No se creará otra orden.",
                    color="green",
                )
                cargar()
                return

            mostrar_snackbar(page, "⏳ Conectando con PayPal...", color="blue")

            tasa = 36.5
            total_usd = round(total / tasa, 2)

            # Reutiliza la orden pendiente para evitar crear varias órdenes
            # cuando el usuario toca el botón más de una vez.
            orden_existente = db.obtener_order_id(reserva_id)
            if orden_existente:
                try:
                    estado_orden = client.execute(OrdersGetRequest(orden_existente)).result
                    if getattr(estado_orden, "status", None) in {"CREATED", "SAVED", "APPROVED"}:
                        approval_link = f"https://www.sandbox.paypal.com/checkoutnow?token={orden_existente}"
                        print(f"ℹ️ Reutilizando orden PayPal existente: {orden_existente}")
                        abrir_url_en_background(page, approval_link)
                        mostrar_snackbar(
                            page,
                            "✅ Reabriendo el pago pendiente de PayPal. Después toca 'Verificar Pago'.",
                            color="green",
                        )
                        return
                except Exception as error_orden:
                    if "RESOURCE_NOT_FOUND" not in str(error_orden) and "INVALID_RESOURCE_ID" not in str(error_orden):
                        raise
                    print(f"⚠️ Orden PayPal inválida; se creará una nueva: {orden_existente}")
                    db.guardar_order_id(reserva_id, None)

            print(f"💳 Creando orden PayPal: reserva #{reserva_id}, C${total} = ${total_usd} USD")

            request = OrdersCreateRequest()
            request.prefer("return=representation")
            request.body = {
                "intent": "CAPTURE",
                "purchase_units": [
                    {
                        "amount": {
                            "currency_code": "USD",
                            "value": str(total_usd),
                        },
                        "payee": {
                            "email_address": paypal_email,
                        },
                        "invoice_id": f"RES-{reserva_id}",
                        "description": f"Reserva #{reserva_id}",
                    }
                ],
                "application_context": {
                    "return_url": "https://example.com/return",
                    "cancel_url": "https://example.com/cancel",
                },
            }

            response = client.execute(request)
            print(f"   Respuesta PayPal: status={response.status_code}")

            if response.status_code == 201:
                order_id = response.result.id
                db.guardar_order_id(reserva_id, order_id)
                print(f"✅ Orden creada en PayPal: {order_id}")

                approval_link = None
                if hasattr(response.result, "links"):
                    for link in response.result.links:
                        if link.rel == "approve":
                            approval_link = link.href
                            break

                if approval_link:
                    print(f"🔗 URL de aprobación: {approval_link}")
                    abrir_url_en_background(page, approval_link)
                    mostrar_snackbar(
                        page,
                        "✅ Abriendo PayPal. Aprobá el pago y luego tocá 'Verificar Pago'.",
                        color="green",
                    )
                else:
                    print("❌ No se obtuvo link de aprobación")
                    mostrar_snackbar(page, "❌ No se obtuvo el enlace de aprobación. Inténtalo nuevamente 😊", color="orange")
            else:
                error_msg = f"Error {response.status_code}"
                if hasattr(response, "result") and hasattr(response.result, "message"):
                    error_msg = response.result.message
                print(f"❌ Error PayPal: {error_msg}")
                mostrar_snackbar(page, ft.Text(f"❌ Error PayPal: {error_msg}", color="orange"))
        except Exception as error:
            print(f"❌ EXCEPCIÓN en procesar_pago_real: {str(error)}")
            import traceback
            traceback.print_exc()
            mostrar_snackbar(page, ft.Text(f"❌ Error: {str(error)}", color="orange"))

                    ),

                spacing=8,

                    "Cancelar",

    def cancelar(rid):
        def confirmar_cancelacion(e):
            db.eliminar_reserva(rid)
            dlg.open = False
            mostrar_snackbar(page, "✅ La reserva fue cancelada y las unidades volvieron al inventario 😊", color="green")
            cargar()

        def cancelar_cancelacion(e):
            dlg.open = False
            page.update()

        dlg = ft.AlertDialog(
            modal=True,
            title=ft.Text("❌ Cancelar Reserva", size=20, weight=ft.FontWeight.BOLD, color="#C62828"),
            content=ft.Text("¿Deseas cancelar esta reserva? Las unidades volverán al inventario 😊", size=14),

                ft.TextButton("No, mantener", on_click=cancelar_cancelacion),
                ft.ElevatedButton(
                    "✅ Eliminar Reserva",
                    on_click=confirmar_cancelacion,
                    bgcolor=ft.Colors.RED_400,
                    color="white",
                ),
            ],
            actions_alignment=ft.MainAxisAlignment.END,
        )
        page.overlay.append(dlg)
        dlg.open = True
        page.update()

    def cargar():
        nonlocal usar_reservas_precargadas
        lista.controls.clear()
        if usar_reservas_precargadas:
            reservas = reservas_precargadas
            usar_reservas_precargadas = False
        else:
            reservas = db.obtener_reservas_usuario(usuario["id"])
        total_c = 0.0
        total_d = 0.0

        for r in reservas:
            rid, nom, cant, pc, pd, estado, tienda_id = r
            subtotal_c = pc * cant
            subtotal_d = pd * cant

            if estado == "Reservado":
                total_c += subtotal_c
                total_d += subtotal_d

            estados_pedido = {
                "Reservado": "⏳ Pendiente de pago (aprueba PayPal y luego verifica)",
                "Pagado": "💳 Pago confirmado; la tienda debe preparar tu pedido.",
                "Preparando": "📦 Tu pedido está siendo preparado.",
                "Listo para entregar": "✅ Tu pedido está listo para entregar.",
                "Entregado": "🎉 Pedido entregado.",
                "Vencido": "⌛ Reserva vencida; el stock fue liberado.",
            }
            estado_mostrar = estados_pedido.get(estado, estado or "Estado no disponible")

            # Tarjeta de reserva: datos, estado actual y acciones disponibles.
            card = ft.Container(
                content=ft.Column(
                    [
                        ft.Text(f"{nom} x{cant}", weight=ft.FontWeight.BOLD),
                        ft.Text(f"C${pc:.2f} c/u = C${subtotal_c:.2f} / ${subtotal_d:.2f}"),
                        ft.Text(f"Estado: {estado_mostrar}", size=13, color="#555555"),
                        ft.Container(
                            content=ft.Row(
                                [
                                    ft.ElevatedButton(
                                        "Pagar con PayPal",
                                        on_click=lambda e, rid=rid, tid=tienda_id, total=subtotal_c: mostrar_pago(rid, tid, total),
                                    ) if estado == "Reservado" else ft.Container(),
                                    ft.ElevatedButton(
                                        "Verificar Pago",
                                        bgcolor="#2E7D32",
                                        color="white",
                                        on_click=lambda e, rid=rid, tid=tienda_id, total=subtotal_c: threading.Thread(
                                            target=ejecutar_captura_paypal,
                                            args=(e, rid, tid, total),
                                            daemon=True,
                                        ).start(),
                                    ) if estado == "Reservado" else ft.Container(),
                                    ft.ElevatedButton(
                                        "Cancelar",
                                        on_click=lambda e, rid=rid: cancelar(rid),
                                        bgcolor=ft.Colors.RED_400,
                                    ) if estado == "Reservado" else ft.Container(),
                                ],
                                spacing=10,
                                wrap=True,
                                scroll=ft.Scrollbar(thickness=4, radius=4, interactive=True),
                            ),
                            width=300,
                        ),
                    ],
                    spacing=5,
                ),
                bgcolor=ft.Colors.WHITE,
                border_radius=15,
                padding=10,
            )
            lista.controls.append(card)

        total_text.value = f"Total: C${total_c:.2f} / ${total_d:.2f}"
        page.update()

    content = ft.Container(
        content=ft.Column(
            [
                ft.Container(height=40),
                ft.Text("Mis Reservas", size=28, weight=ft.FontWeight.BOLD, color="#040D4A"),
                ft.Text("Las reservas expiran en 24 horas", size=14, weight=ft.FontWeight.BOLD, color="#0A2540", italic=True),

    def normalizar_datos_imagen(image_data):
        if image_data is None:
            return None
        if isinstance(image_data, str):
            try:
                return base64.b64decode(image_data.split(",", 1)[-1], validate=True)
            except (ValueError, base64.binascii.Error):
                return None
        try:
            datos = bytes(image_data)
        except (TypeError, ValueError):
            return None
        return datos or None

    def obtener_fuente_imagen(ruta, image_data=None):

        return {"src": ruta}

            with open(ruta, "wb") as f:
                f.write(image_data)

def _obtener_ruta_sesion():
    almacenamiento_app = os.environ.get("FLET_APP_STORAGE_DATA")
    if almacenamiento_app:
        directorio = Path(almacenamiento_app)
        directorio.mkdir(parents=True, exist_ok=True)
        return directorio / "sesion.json"
    return Path(__file__).resolve().parent.parent / "sesion.json"


SESION_FILE = _obtener_ruta_sesion()
SESION_KEY = "smartstore_session"


def generar_codigo_sesion(longitud=None):
    if longitud is None:
        longitud = secrets.randbelow(5) + 12
    if longitud < 3:
        raise ValueError("El código de sesión debe tener al menos 3 caracteres")

    letras = string.ascii_letters
    numeros = string.digits
    simbolos = "#*/?&%$@sebnmotyuacnq.iwxz1234567890)+,!:;=<"
    caracteres = letras + numeros + simbolos
    codigo = [
        secrets.choice(letras),
        secrets.choice(numeros),
        secrets.choice(simbolos),
    ]
    codigo.extend(secrets.choice(caracteres) for _ in range(longitud - len(codigo)))
    secrets.SystemRandom().shuffle(codigo)
    return "".join(codigo)


def invalidar_codigo_sesion(page):
    page._smartstore_session_code = None


def invalidar_codigo_recuperacion(page):
    page._smartstore_recovery_code = None


def codigo_sesion_valido(page, codigo):
    esperado = getattr(page, "_smartstore_session_code", None)
    return bool(esperado and codigo and secrets.compare_digest(str(esperado), str(codigo)))


def codigo_recuperacion_valido(page, codigo):
    esperado = getattr(page, "_smartstore_recovery_code", None)
    return bool(esperado and codigo and secrets.compare_digest(str(esperado), str(codigo)))


def guardar_sesion(page, correo, password, usuario_id=None):
    correo = (correo or "").strip()
    password = (password or "").strip()

    if not correo or (not password and usuario_id is None):
        return

    token = db.crear_token_sesion(correo, password, usuario_id=usuario_id)
    if not token:
        return
    auth_refresh_token = obtener_refresh_token_auth()

    try:
        page.client_storage.set(SESION_KEY, {"token": token, "auth_refresh_token": auth_refresh_token})
        return
    except Exception:
        pass
    try:
        with open(SESION_FILE, "w", encoding="utf-8") as f:
            json.dump({"token": token, "auth_refresh_token": auth_refresh_token}, f)
    except Exception:
        pass


def cargar_sesion(page):
    try:
        datos = page.client_storage.get(SESION_KEY)
        if datos:
            token = str(datos.get("token") or "").strip()
            refresh_token = datos.get("auth_refresh_token")
            db.restaurar_sesion_auth(refresh_token)
            nuevo_refresh_token = obtener_refresh_token_auth()
            usuario = db.obtener_usuario_por_token(token)
            if usuario:
                if nuevo_refresh_token and nuevo_refresh_token != refresh_token:
                    datos["auth_refresh_token"] = nuevo_refresh_token
                    page.client_storage.set(SESION_KEY, datos)
                return usuario[2], token
            borrar_sesion(page)
            return None, None
    except Exception:
        pass
    try:
        if SESION_FILE.exists():
            with open(SESION_FILE, "r", encoding="utf-8") as f:
                datos = json.load(f)
            token = str(datos.get("token") or "").strip()
            refresh_token = datos.get("auth_refresh_token")
            db.restaurar_sesion_auth(refresh_token)
            nuevo_refresh_token = obtener_refresh_token_auth()
            usuario = db.obtener_usuario_por_token(token)
            if usuario:
                if nuevo_refresh_token and nuevo_refresh_token != refresh_token:
                    datos["auth_refresh_token"] = nuevo_refresh_token
                    with open(SESION_FILE, "w", encoding="utf-8") as f:
                        json.dump(datos, f)
                return usuario[2], token
            borrar_sesion(page)
            return None, None
    except Exception:
        pass
    return None, None


def borrar_sesion(page):
    invalidar_codigo_sesion(page)
    try:
        datos = page.client_storage.get(SESION_KEY)
        db.revocar_token_sesion((datos or {}).get("token"))
    except Exception:
        pass
    try:
        page.client_storage.remove(SESION_KEY)
    except Exception:
        pass
    try:
        if SESION_FILE.exists():
            try:
                with open(SESION_FILE, "r", encoding="utf-8") as f:
                    datos = json.load(f)
                db.revocar_token_sesion((datos or {}).get("token"))
            except Exception:
                pass
            SESION_FILE.unlink()
    except Exception:
        pass


def iniciar_google_oauth(page, cuando_exito, cuando_error):
    """Abre Google y devuelve el resultado OAuth a la ventana de escritorio."""
    verifier = secrets.token_urlsafe(48)
    challenge = base64.urlsafe_b64encode(
        hashlib.sha256(verifier.encode("ascii")).digest()
    ).rstrip(b"=").decode("ascii")
    redirect_to = "http://127.0.0.1:8765/auth/callback"

    class CallbackHandler(BaseHTTPRequestHandler):
        def do_GET(self):
            from urllib.parse import parse_qs, urlparse
            parametros = parse_qs(urlparse(self.path).query)
            codigo = parametros.get("code", [None])[0]
            error = parametros.get("error_description", [None])[0]
            self.send_response(200)
            self.send_header("Content-Type", "text/html; charset=utf-8")
            self.end_headers()
            self.wfile.write(b"<h2>Puedes volver a SmartStore.</h2>")
            self.server.resultado = (codigo, error)

            def detener():
                self.server.shutdown()

            threading.Thread(target=detener, daemon=True).start()

        def log_message(self, *_args):
            return

    try:
        servidor = HTTPServer(("127.0.0.1", 8765), CallbackHandler)
    except OSError:
        cuando_error("El puerto OAuth local está ocupado. Cierra otra instancia e inténtalo nuevamente.")
        return
    servidor.resultado = (None, None)
    url = crear_url_oauth_google(redirect_to, challenge)

    def esperar_callback():
        servidor.serve_forever()
        codigo, error = servidor.resultado
        servidor.server_close()
        if not codigo:
            cuando_error(error or "Google no completó la autenticación.")
            return
        try:
            respuesta = intercambiar_codigo_oauth(codigo, verifier)
            cuando_exito(respuesta)
        except Exception as exc:
            cuando_error(str(exc))

    threading.Thread(target=esperar_callback, daemon=True).start()

    async def abrir_navegador():
        try:
            resultado = page.launch_url(url)
            if hasattr(resultado, "__await__"):
                await resultado
        except Exception as exc:
            cuando_error(f"No se pudo abrir el navegador: {exc}")

    page.run_task(abrir_navegador)

        try:
            fecha_nac = datetime.strptime(fecha_nac_str, "%Y-%m-%d")
            hoy = datetime.now()
            edad = hoy.year - fecha_nac.year
            if (hoy.month, hoy.day) < (fecha_nac.month, fecha_nac.day):
                edad -= 1
            return edad
        except Exception:
            return -1

    def accion(e):
        if es_registro:
            nombre = nombre_input.value.strip()
            correo = correo_input.value.strip()
            telefono = telefono_input.value.strip()
            fecha_str = fecha_input.value.strip()
            password = password_input.value

            if not nombre or not correo or not telefono or not fecha_str or (not password and not usuario_google):
                mostrar_mensaje("Por favor, completa todos los campos para continuar. 😊")
                return

            if not usuario_google and len(password) < 6:
                mostrar_mensaje("La contraseña debe tener al menos 6 caracteres.")
                return

            edad = calcular_edad(fecha_str)
            if edad == -1:
                mostrar_mensaje("⚠️ La fecha no tiene un formato válido. Usa AAAA-MM-DD, por ejemplo: 2009-05-15 😊")
                return

            if edad < 17:
                mostrar_mensaje(f"⚠️ Debes tener al menos 17 años para registrarte. Actualmente tienes {edad} años 😊")
                return

            imagen_rel = guardar_imagen_en_assets(imagen_perfil_path, prefijo="perfil") if imagen_perfil_path else None
            try:
                if usuario_google:
                    if len(password) < 6:
                        mostrar_mensaje("La contraseña SmartStore debe tener al menos 6 caracteres.")
                        return
                    if not db.establecer_contraseña_auth(password):
                        panel_codigo.visible = False
                        mostrar_mensaje("❌ No se pudo crear la contraseña de SmartStore.")
                        return
                    exitoso, mensaje = db.crear_perfil_auth(
                        usuario_google.get("id"), nombre, correo, telefono, fecha_str, imagen_rel
                    )
                elif db._supabase_activo():
                    exitoso, mensaje = db.registrar_usuario_auth(
                        nombre, correo, telefono, fecha_str, password, imagen_rel
                    )
                else:
                    exitoso, mensaje = db.registrar_usuario(
                        nombre, correo, telefono, fecha_str, password, imagen_rel
                    )
            except Exception as e:
                log_error(f"Error registrar usuario: {e}")
                mostrar_snackbar(page, f"❌ Error al registrar usuario: {e}", color="red")
                return
            mostrar_mensaje(mensaje)

            if exitoso:
                if usuario_google:
                    usuario = db.obtener_usuario_por_correo(correo)
                    if not usuario:
                        mostrar_mensaje("❌ No se pudo preparar tu perfil.")
                        return
                    usuario_dict = {"id": usuario[0], "nombre": usuario[1], "correo": usuario[2]}
                    mostrar_codigo_y_continuar(usuario_dict, correo, "")
                    return
                if "confirma tu correo" in mensaje.lower():
                    return
                try:
                    usuario = db.obtener_usuario(correo, password)
                except Exception as error:
                    log_error(f"Error al preparar el código de la cuenta nueva: {error}")
                    mostrar_snackbar(page, "✅ Cuenta creada. Inicia sesión para recibir tu código de seguridad.", color="green")
                    correo_input.value = correo
                    password_input.value = password
                    cambiar_vista(limpiar_credenciales=False)
                    return

                if not usuario:
                    mostrar_snackbar(page, "✅ Cuenta creada. Inicia sesión para recibir tu código de seguridad.", color="green")
                    correo_input.value = correo
                    password_input.value = password
                    cambiar_vista(limpiar_credenciales=False)
                    return

                usuario_dict = {"id": usuario[0], "nombre": usuario[1], "correo": usuario[2]}
                mostrar_codigo_y_continuar(usuario_dict, correo, password)
        else:
            correo = correo_input.value.strip()
            password = password_input.value

            if not correo or not password:
                mostrar_mensaje("🔐 Ingresa tu correo y contraseña para continuar 😊")
                return

            try:
                db.verificar_conexion_remota()
                mostrar_spinner_temporal()
            except RuntimeError as exc:
                ocultar_spinner_reconexion()
                if "desactivado" in str(exc).lower():
                    mostrar_mensaje("⚙️ Supabase está desactivado. Cierra y vuelve a abrir la app con SMARTSTORE_USAR_SUPABASE=1.")
                elif "no está configurado" in str(exc).lower():
                    mostrar_mensaje("⚙️ Supabase no está configurado correctamente. Revisa la URL y la clave anon/public.")
                else:
                    mostrar_mensaje("⚠️ No se pudo conectar con Supabase. Revisa la conexión y las políticas RLS.")
                return

            try:
                if db._supabase_activo():
                    resultado_auth = db.iniciar_sesion_auth(correo, password)
                    usuario = resultado_auth[2] if resultado_auth else None
                else:
                    usuario = db.obtener_usuario(correo, password)
            except RuntimeError:
                ocultar_spinner_reconexion()
                mostrar_mensaje("⚠️ Parece que no tienes conexión a internet 😅 Revisa tu red y vuelve a intentarlo cuando estés listo 😊")
                return

            if usuario:
                usuario_dict = {"id": usuario[0], "nombre": usuario[1], "correo": usuario[2]}
                mostrar_codigo_y_continuar(usuario_dict, correo, password)
            else:
                mostrar_mensaje("❌ El correo o la contraseña no coinciden 😅 Revisa los datos y prueba otra vez, aquí estamos para ayudarte 😊")

    btn_accion.on_click = accion
    btn_alternar.on_click = toggle
    correo_input.on_submit = accion
    password_input.on_submit = accion
    nombre_input.on_submit = accion

        ancho_disponible = min(340, max(190, (page.width or 420) - 40))
        for campo in (correo_input, password_input, nombre_input, telefono_input, fecha_input):

    page.add(
        ft.Stack(
            [
                fondo,
                ft.SafeArea(form),
            ],
            expand=True,
        )
    )

    actualizar_ancho_responsive()

import flet as ft
import asyncio
import time
from utils import configurar_retroceso

def pre_splash_view(page: ft.Page, on_complete):
    # PANTALLA: carga inicial de la aplicación y alineación general.
    configurar_retroceso(page, lambda: None)
    page.clean()
    page.padding = 0
    page.spacing = 0
    page.margin = 0
    page.bgcolor = "#EF7620"
    page.update()

    progress = ft.ProgressRing(
        width=40,       # Ancho del circulito en píxeles.
        height=40,      # Alto del circulito en píxeles.
        stroke_width=5, # Grosor de la línea del circulito en píxeles.
        color="#040D4A",
    )

    # CONTROLES: identidad visual de inicio e indicador de precarga de datos.
    content = ft.Column(
        [
            ft.Text("😉", size=80),
            ft.Text("🛒", size=80),
            ft.Text("¡Smart-Store!", size=32, weight=ft.FontWeight.BOLD, color="#FFFFFF"),
            ft.Container(height=6), #espaciado entre el título y el indicador de progreso.
            progress,
            ft.Text("Cargando datos...", size=15, color="#FFFFFF"),
        ],
        horizontal_alignment=ft.CrossAxisAlignment.CENTER,
        spacing=20,
        alignment=ft.MainAxisAlignment.CENTER,
        expand=True,
    )
    
    # MONTAJE: agrega la pantalla inicial a la página.
    page.add(content)
    page.update()

    async def animar_progreso():
        """Mantiene el indicador visible al menos 6 segundos y luego avanza."""
        inicio = time.monotonic()
        try:
            await asyncio.wait_for(page._smartstore_start_ready.wait(), timeout=30)
        except asyncio.TimeoutError:
            # La interfaz puede continuar aunque el servidor tarde demasiado.
            pass

        transcurrido = time.monotonic() - inicio
        if transcurrido < 6:
            await asyncio.sleep(6 - transcurrido)

        page.clean()
        page.update()
        on_complete()

    page.run_task(animar_progreso)

    def normalizar_datos_imagen(image_data):
        if image_data is None:
            return None
        if isinstance(image_data, str):
            try:
                return base64.b64decode(image_data.split(",", 1)[-1], validate=True)
            except (ValueError, base64.binascii.Error):
                return None
        try:
            datos = bytes(image_data)
        except (TypeError, ValueError):
            return None
        return datos or None

    def obtener_fuente_imagen(ruta, image_data=None):
        datos = normalizar_datos_imagen(image_data)
        if datos:
            return {"src": base64.b64encode(datos).decode("ascii")}
        return {"src": ruta}

import flet as ft
import asyncio
from utils import configurar_retroceso
import hashlib
contrasena = hashlib.md5(b"mi_clave_secreta") 
def splash_view(page: ft.Page, on_complete):
    # PANTALLA: bienvenida de la aplicación a pantalla completa.
    configurar_retroceso(page, lambda: None)
    page.clean()
    page.update()
    page.bgcolor = "#F5F7FA"
    page.padding = 0
    page.spacing = 0
    page.margin = 0

    fondo = ft.Image(
        src="Bienvenida.png",
        fit="cover",
        expand=True,
    )

    # BOTÓN: permite avanzar desde la bienvenida hacia la siguiente pantalla.
    btn_avanzar = ft.ElevatedButton(
    content=ft.Column(
        [
            ft.Text("📎", size=18),
            ft.Text("🙃", size=16),
        ],
        alignment=ft.MainAxisAlignment.CENTER,
        horizontal_alignment=ft.CrossAxisAlignment.CENTER,
        tight=True,
    ),
    bgcolor="#C3D2D6",
    color="white",
    width=60,
    height=60,
    style=ft.ButtonStyle(
        shape=ft.RoundedRectangleBorder(radius=5)
    )
)

    finalizado = {"valor": False}

    def avanzar(e=None):
        # ACCIÓN: limpia la bienvenida y continúa con el flujo de inicio.
        if finalizado["valor"]:
            return
        finalizado["valor"] = True
        page.clean()
        page.update()
        on_complete()

    async def auto_avanzar():
        await asyncio.sleep(6)
        if not finalizado["valor"]:
            avanzar()

    page.run_task(auto_avanzar)

    btn_avanzar.on_click = avanzar

    # LAYOUT: coloca el botón de entrada abajo y a la derecha.
    capa_contenido = ft.Container(
        content=ft.Column(
            [btn_avanzar],
            alignment=ft.MainAxisAlignment.END,          # Botón abajo
            horizontal_alignment=ft.CrossAxisAlignment.END,  # Botón derecha
        ),
        expand=True,
        padding=30,  # Margen desde los bordes
    )

    # MONTAJE: superpone la imagen de bienvenida y el botón de entrada.
    stack = ft.Stack(
        [
            fondo,
            capa_contenido,
        ],
        expand=True
    )

    page.add(stack)
    page.update()

import flet as ft
import database as db
from utils import mostrar_snackbar, configurar_retroceso


def top_productos_view(page: ft.Page, usuario, go_back):
    # PANTALLA: ranking de productos más comprados y navegación de regreso.
    # Pantalla de productos destacados según compras pagadas.
    configurar_retroceso(page, lambda: go_back(usuario))
    page.clean()
    page.bgcolor = "white"

    fondo = ft.Container(
        expand=True,
        image=ft.DecorationImage(src="fondo.png", fit="cover"),
    )

    productos = db.obtener_top_productos(limit=10, min_compras=3)

    # Lista de tarjetas con nombre, precios, compras y popularidad.
    lista = ft.Column(spacing=10)

    if not productos:
        lista.controls.append(ft.Text("Aún no hay productos con al menos 3 compras pagadas 😊", width=300, text_align=ft.TextAlign.CENTER))
    else:
        for p in productos:
            # p: id, nombre, precio_cordobas, precio_dolares, stock, compras, promedio_calificacion
            pid = p[0]
            nombre = p[1]
            precio_c = float(p[2]) if p[2] else 0
            precio_d = float(p[3]) if p[3] else 0
            stock = int(p[4]) if p[4] else 0
            compras = int(p[5]) if p[5] else 0
            # La clasificación del top refleja popularidad, no reseñas manuales:
            # 3 compras = 1 estrella; cada 2 compras adicionales suma una estrella.
            estrellas_llenas = 1 if compras == 3 else min(5, ((compras - 4) // 2) + 2)
            estrellas = "★" * estrellas_llenas + "☆" * (5 - estrellas_llenas)
            rating_text = f"{estrellas}  Popularidad: {estrellas_llenas}/5"

            card = ft.Container(
                content=ft.Row(
                    [
                        ft.Column([
                            ft.Text(nombre, weight=ft.FontWeight.BOLD),
                            ft.Text(f"C${precio_c:.2f} / ${precio_d:.2f}"),
                            ft.Text(f"Compras: {compras}"),
                            ft.Text(rating_text, color="#EFA520", weight=ft.FontWeight.BOLD),
                        ]),
                    ],
                ),
                bgcolor=ft.Colors.WHITE,
                border_radius=12,
                padding=10,
            )
            lista.controls.append(card)

    content = ft.Container(
        content=ft.Column(
            [
                ft.Container(height=40),
                ft.Text("Top Productos", size=28, weight=ft.FontWeight.BOLD, color="#040D4A"),
                lista,
                ft.ElevatedButton("↩️ Volver", on_click=lambda e: go_back(usuario)),
            ],
            horizontal_alignment=ft.CrossAxisAlignment.CENTER,
            alignment=ft.MainAxisAlignment.CENTER,
            spacing=20,
            scroll=ft.Scrollbar(thickness=4, radius=4, interactive=True),
        ),
        padding=20,
    )

    page.add(
        ft.Stack(
            [
                fondo,
                ft.SafeArea(content)
            ],
            expand=True
        )
    )
