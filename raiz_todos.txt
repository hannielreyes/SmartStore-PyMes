"""Estado de datos de la sesión actual y precarga desde Supabase."""

from concurrent.futures import ThreadPoolExecutor
from datetime import datetime

import database as db


def cargar_contexto_usuario(usuario_id):
    """Carga el contexto inicial sin bloquear el hilo de interfaz de Flet."""
    tiendas = db.obtener_tiendas_usuario(usuario_id)
    tienda_ids = [tienda[0] for tienda in tiendas]

    def cargar_productos():
        return db.obtener_productos_usuario(usuario_id)

    def cargar_paypal():
        return db.obtener_paypal_email(usuario_id)

    def cargar_reservas():
        reservas = []
        for tienda_id in tienda_ids:
            try:
                reservas.extend(db.obtener_reservas_por_tienda(tienda_id))
            except Exception:
                continue
        return reservas

    def cargar_reservas_usuario():
        return db.obtener_reservas_usuario(usuario_id)

    with ThreadPoolExecutor(max_workers=3) as executor:
        futuros = {
            "productos": executor.submit(cargar_productos),
            "paypal": executor.submit(cargar_paypal),
            "reservas": executor.submit(cargar_reservas),
            "reservas_usuario": executor.submit(cargar_reservas_usuario),
        }
        productos = futuros["productos"].result()
        paypal = futuros["paypal"].result()
        reservas = futuros["reservas"].result()
        reservas_usuario = futuros["reservas_usuario"].result()

    alertas = [
        (producto[0], producto[1], int(producto[6] or 0))
        for producto in productos
        if int(producto[6] or 0) == 1
    ][:20]

    return {
        "usuario_id": usuario_id,
        "tiendas": tiendas,
        "tienda_actual": tiendas[0] if tiendas else None,
        "productos": productos,
        "stock_total": sum(int(producto[6] or 0) for producto in productos),
        "alertas_inventario": alertas,
        "reservas": reservas,
        "reservas_usuario": reservas_usuario,
        "paypal": paypal,
        "actualizado_en": datetime.now(),
    }

def _cargar_config_build_temporal():
    """Lee la config de build si existe; evita warnings de importación cuando no existe."""
    ruta = os.path.join(os.path.dirname(__file__), "supabase_build_config.py")
    if not os.path.exists(ruta):
        return "", "", "", ""

    spec = importlib.util.spec_from_file_location("supabase_build_config", ruta)
    if spec is None or spec.loader is None:
        return "", "", "", ""

    modulo = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(modulo)
    return (
        getattr(modulo, "SUPABASE_URL", ""),
        getattr(modulo, "SUPABASE_KEY", ""),
        getattr(modulo, "UPDATE_URL", ""),
        getattr(modulo, "APP_VERSION", ""),
    )

def _cargar_env_local():
    """Carga .env desde la raiz del proyecto y otros ubicaciones comunes."""
    directorios = []
    base_dir = os.path.dirname(__file__)
    for ruta in (base_dir, os.path.dirname(base_dir), os.getcwd()):
        if ruta and ruta not in directorios:
            directorios.append(ruta)

    rutas_env = []
    for directorio in directorios:
        for nombre_archivo in (".env", ".env.local"):
            ruta = os.path.join(directorio, nombre_archivo)
            if os.path.exists(ruta) and ruta not in rutas_env:
                rutas_env.append(ruta)

    for ruta_env in rutas_env:
        try:
            with open(ruta_env, "r", encoding="utf-8") as archivo:
                for linea in archivo:
                    linea = linea.strip()
                    if not linea or linea.startswith("#") or "=" not in linea:
                        continue
                    nombre, valor = linea.split("=", 1)
                    nombre = nombre.strip()
                    valor = valor.strip()
                    if len(valor) >= 2 and valor[0] == valor[-1] and valor[0] in {'"', "'"}:
                        valor = valor[1:-1]
                    os.environ.setdefault(nombre, valor)
        except OSError:
            continue

def _valor_es_placeholder(valor):
    if not valor:
        return True

    valor_limpio = valor.strip().lower()
    placeholders = (
        "tu-proyecto",
        "tu_proyecto",
        "tu-clave",
        "tu_clave_anon_public",
        "tu_clave_anon",
        "tu_key",
        "tu-llave",
        "your-project",
        "your_project",
        "your-key",
        "your_key",
        "example.com",
        "business.example.com",
    )

    if valor_limpio in placeholders:
        return True

    if valor_limpio.startswith("https://tu-proyecto") or valor_limpio.startswith("http://tu-proyecto"):
        return True

    if "tu_proyecto" in valor_limpio or "tu-proyecto" in valor_limpio:
        return True

    if "tu_clave" in valor_limpio or "tu-key" in valor_limpio or "tu_llave" in valor_limpio:
        return True

    return False


def supabase_esta_configurado():
    return (
        bool(SUPABASE_URL)
        and bool(SUPABASE_KEY)
        and not _valor_es_placeholder(SUPABASE_URL)
        and not _valor_es_placeholder(SUPABASE_KEY)
        and not _clave_es_service_role(SUPABASE_KEY)
    )


def _clave_es_service_role(clave):
    """Detecta una JWT service_role antes de usarla en el cliente APK."""
    partes = str(clave or "").split(".")
    if len(partes) != 3:
        return False
    try:
        relleno = "=" * (-len(partes[1]) % 4)
        payload = json.loads(base64.urlsafe_b64decode(partes[1] + relleno))
    except (ValueError, TypeError, json.JSONDecodeError):
        return False
    return payload.get("role") == "service_role"


def _headers():
    if not SUPABASE_KEY:
        raise RuntimeError(
            "Falta SMARTSTORE_SUPABASE_KEY. Configura la clave anon/public "
            "antes de probar Supabase."
        )
    if _clave_es_service_role(SUPABASE_KEY):
        raise RuntimeError(
            "SMARTSTORE_SUPABASE_KEY es service_role. Usa la clave anon/public en la APK."
        )
    return {
        "apikey": SUPABASE_KEY,
        "Authorization": f"Bearer {_SUPABASE_ACCESS_TOKEN or SUPABASE_KEY}",
        "Content-Type": "application/json",
        "Prefer": "return=representation",
    }


def establecer_token_auth(access_token):
    """Usa el JWT del usuario para que Supabase aplique RLS por identidad."""
    global _SUPABASE_ACCESS_TOKEN
    _SUPABASE_ACCESS_TOKEN = str(access_token or "").strip() or None


def establecer_sesion_auth(access_token, refresh_token=None):
    global _SUPABASE_ACCESS_TOKEN, _SUPABASE_REFRESH_TOKEN
    _SUPABASE_ACCESS_TOKEN = str(access_token or "").strip() or None
    _SUPABASE_REFRESH_TOKEN = str(refresh_token or "").strip() or None


def obtener_refresh_token_auth():
    return _SUPABASE_REFRESH_TOKEN


def refrescar_sesion_auth(refresh_token):
    global _SUPABASE_ACCESS_TOKEN, _SUPABASE_REFRESH_TOKEN
    _, respuesta = supabase_auth_request(
        "token?grant_type=refresh_token",
        datos={"refresh_token": refresh_token},
    )
    access_token = respuesta.get("access_token") if isinstance(respuesta, dict) else None
    nuevo_refresh = respuesta.get("refresh_token") if isinstance(respuesta, dict) else None
    if not access_token:
        limpiar_token_auth()
        return False
    _SUPABASE_ACCESS_TOKEN = access_token
    _SUPABASE_REFRESH_TOKEN = nuevo_refresh or refresh_token
    return True


def limpiar_token_auth():
    global _SUPABASE_REFRESH_TOKEN
    global _SUPABASE_ACCESS_TOKEN
    _SUPABASE_ACCESS_TOKEN = None
    _SUPABASE_REFRESH_TOKEN = None


def supabase_auth_request(ruta, metodo="POST", datos=None):
    """Llama a GoTrue usando siempre la clave pública de la aplicación."""
    url = quote(f"{SUPABASE_URL}/auth/v1/{ruta}", safe=":/?&=,*().-_'%")
    cuerpo = None if datos is None else json.dumps(datos).encode("utf-8")
    request = Request(url, data=cuerpo, headers={
        "apikey": SUPABASE_KEY,
        "Authorization": f"Bearer {_SUPABASE_ACCESS_TOKEN or SUPABASE_KEY}",
        "Content-Type": "application/json",
    }, method=metodo.upper())
    try:
        with urlopen(request, timeout=8) as response:
            contenido = response.read().decode("utf-8")
            return response.status, json.loads(contenido) if contenido else None
    except HTTPError as error:
        detalle = error.read().decode("utf-8", errors="replace")
        raise RuntimeError(f"Supabase Auth respondió HTTP {error.code}: {detalle}") from error
    except URLError as error:
        raise RuntimeError(f"No se pudo contactar Supabase Auth: {error.reason}") from error


def crear_url_oauth_google(redirect_to, code_challenge):
    parametros = (
        f"?provider=google&redirect_to={quote(str(redirect_to), safe='') }"
        f"&code_challenge={quote(str(code_challenge), safe='')}&code_challenge_method=S256"
    )
    return f"{SUPABASE_URL}/auth/v1/authorize{parametros}"


def intercambiar_codigo_oauth(codigo, verifier):
    _, respuesta = supabase_auth_request(
        "token?grant_type=pkce",
        datos={"auth_code": codigo, "code_verifier": verifier},
    )
    if not isinstance(respuesta, dict) or not respuesta.get("access_token"):
        raise RuntimeError("Supabase no devolvió una sesión OAuth válida.")
    establecer_sesion_auth(respuesta["access_token"], respuesta.get("refresh_token"))
    return respuesta


def comprobar_conexion():
    """Comprueba que Supabase responde con la clave publica configurada."""
    request = Request(f"{SUPABASE_URL}/auth/v1/settings", headers=_headers(), method="GET")
    try:
        with urlopen(request, timeout=8) as response:
            contenido = response.read().decode("utf-8")
            return response.status, contenido
    except HTTPError as error:
        detalle = error.read().decode("utf-8", errors="replace")
        raise RuntimeError(f"Supabase respondio HTTP {error.code}: {detalle}") from error
    except URLError as error:
        raise RuntimeError(f"No se pudo contactar Supabase: {error.reason}") from error


def supabase_request(tabla, metodo="GET", datos=None, parametros=""):
    """Realiza una peticion REST basica; la API central usara esta funcion."""
    url = f"{SUPABASE_URL}/rest/v1/{tabla}{parametros}"
    url = quote(url, safe=":/?&=,*().-_'%")
    cuerpo = None if datos is None else json.dumps(datos).encode("utf-8")

    ultimo_error = None
    for intento in range(3):
        request = Request(url, data=cuerpo, headers=_headers(), method=metodo.upper())
        try:
            with urlopen(request, timeout=8) as response:
                contenido = response.read().decode("utf-8")
                return response.status, json.loads(contenido) if contenido else None
        except HTTPError as error:
            detalle = error.read().decode("utf-8", errors="replace")
            ultimo_error = RuntimeError(f"Supabase respondio HTTP {error.code}: {detalle}")
            if error.code in (408, 429, 500, 502, 503, 504) and intento < 2:
                continue
            raise ultimo_error from error
        except URLError as error:
            ultimo_error = RuntimeError(f"No se pudo contactar Supabase: {error.reason}")
            if intento < 2:
                continue
            raise ultimo_error from error

    if ultimo_error is not None:
        raise ultimo_error
    raise RuntimeError("Supabase respondió sin datos válidos.")


def supabase_rpc(nombre, datos=None):
    """Invoca una funcion PostgreSQL publicada como RPC en Supabase."""
    _, resultado = supabase_request(
        f"rpc/{nombre}", metodo="POST", datos=datos or {}
    )
    return resultado


def subir_archivo_storage(ruta_archivo, carpeta="uploads"):
    """Sube una imagen al bucket publico compartido y devuelve su URL."""
    if not ruta_archivo or not os.path.exists(ruta_archivo) or not SUPABASE_URL or not SUPABASE_KEY:
        return None

    nombre = f"{carpeta}/{uuid.uuid4().hex}{os.path.splitext(ruta_archivo)[1].lower() or '.jpg'}"
    url = f"{SUPABASE_URL}/storage/v1/object/smartstore-media/{nombre}"
    tipo, _ = mimetypes.guess_type(ruta_archivo)
    headers = _headers()
    headers["Content-Type"] = tipo or "application/octet-stream"
    headers["x-upsert"] = "true"

    with open(ruta_archivo, "rb") as archivo:
        request = Request(url, data=archivo.read(), headers=headers, method="POST")
    try:
        with urlopen(request, timeout=30):
            return f"{SUPABASE_URL}/storage/v1/object/public/smartstore-media/{nombre}"
    except (HTTPError, URLError):
        return None

def _es_android(page: ft.Page) -> bool:
    """Detecta si la app está corriendo en Android (APK)."""
    try:
        plataforma = getattr(page, "platform", None)
        if plataforma is not None:
            valor = str(plataforma).lower()
            if valor in {"android", "android_tv", "android-tv"}:
                return True
            if "android" in valor:
                return True
    except Exception:
        pass

    try:
        import platform
        sistema = str(platform.system()).lower()
        return sistema == "android"
    except Exception:
        return False

def notificaciones_activas(page):
    """Devuelve si el usuario permite avisos informativos dentro de la app."""
    try:
        with _ruta_preferencias_notificaciones().open("r", encoding="utf-8") as archivo:

    except Exception:
        pass

    try:
        valor = page.client_storage.get(NOTIFICACIONES_ACTIVAS_KEY)
        if valor is not None:
            return bool(valor)
    except Exception:
        pass
    return getattr(page, "_notificaciones_activas", True)

    page._notificaciones_activas = activas
    try:
        page.client_storage.set(NOTIFICACIONES_ACTIVAS_KEY, activas)
    except Exception:
        pass
    try:
        with _ruta_preferencias_notificaciones().open("w", encoding="utf-8") as archivo:
            json.dump({NOTIFICACIONES_ACTIVAS_KEY: activas}, archivo)
    except OSError:
        pass
    return activas

def configurar_retroceso(page: ft.Page, callback):
    """Conecta el botón o gesto de retroceso del teléfono con la vista actual.
    """
    # Variable global para guardar el callback de retroceso
    global _retroceso_callback
    _retroceso_callback = callback

    retroceso_en_proceso = False

    def manejar_retroceso(e=None):
        nonlocal retroceso_en_proceso
        if retroceso_en_proceso:
            return
        retroceso_en_proceso = True
        try:
            if _retroceso_callback is not None:
                _retroceso_callback()
        finally:
            retroceso_en_proceso = False

    def manejar_tecla(e):
        tecla = str(getattr(e, "key", "")).strip().lower()
        if tecla in {"goback", "browserback", "escape", "back"}:
            try:
                if hasattr(e, "prevent_default"):
                    e.prevent_default()
            except Exception:
                pass
            manejar_retroceso(e)

    page.on_view_pop = manejar_retroceso
    page.on_keyboard_event = manejar_tecla

    try:
        if hasattr(page, "window"):
            plataforma = getattr(page, "platform", None)
            if plataforma in (ft.PagePlatform.ANDROID, ft.PagePlatform.ANDROID_TV, ft.PagePlatform.IOS):
                page.window.prevent_close = True
            else:
                page.window.prevent_close = False
    except Exception:
        pass

def guardar_imagen_en_assets(origen: str | None, prefijo: str = "img"):
    if not origen:
        return None

    # Si ya viene como URL o data URI, no hay que volver a procesarla.
    if isinstance(origen, str) and origen.startswith(("http://", "https://", "data:")):
        return origen

    if not os.path.exists(origen):
        return None

    base_dir = _obtener_directorio_imagenes()
    uploads_dir = base_dir / "uploads"
    uploads_dir.mkdir(parents=True, exist_ok=True)

    extension = Path(origen).suffix.lower() or ".png"
    nombre_archivo = f"{prefijo}_{uuid.uuid4().hex}{extension}"
    destino = uploads_dir / nombre_archivo
    shutil.copy2(origen, destino)

    ruta_local = str(destino.resolve())
    try:
        from supabase_config import subir_archivo_storage
        url_remota = subir_archivo_storage(ruta_local, carpeta=prefijo)
        if url_remota:
            return url_remota
    except Exception as error:
        log_error(f"No se pudo subir imagen a Supabase Storage: {error}", exc_info=False)

    try:
        tipo = mimetypes.guess_type(ruta_local)[0] or "image/jpeg"
        with open(ruta_local, "rb") as archivo:
            contenido = base64.b64encode(archivo.read()).decode("ascii")
        return f"data:{tipo};base64,{contenido}"
    except OSError as error:
        log_error(f"No se pudo preparar imagen compartida: {error}", exc_info=False)
    return ruta_local

def normalizar_whatsapp(whatsapp: str | None):
    """Normaliza un número de WhatsApp para usarlo en URLs y app links."""
    if whatsapp is None:
        return None

    texto = str(whatsapp).strip()
    texto = re.sub(r"[^0-9+]", "", texto)

    if not texto:
        return None

    if texto.startswith("+"):
        texto = texto[1:]

    if len(texto) < 8 or not texto.isdigit():
        return None

    if len(texto) == 8:
        texto = f"505{texto}"

    return texto


def construir_url_whatsapp(whatsapp: str | None, mensaje: str = ""):
    numero = normalizar_whatsapp(whatsapp)
    if not numero:
        return None

    url = f"https://wa.me/{numero}"
    texto = (mensaje or "").strip()
    if texto:
        url += f"?text={quote(texto)}"
    return url

def abrir_whatsapp(page: ft.Page, whatsapp: str | None, mensaje: str = "") -> bool:
    """Confirma la intención del cliente y abre el chat de la tienda."""
    if not page:
        return False

    numero = normalizar_whatsapp(whatsapp)
    if not numero:
        return False

    texto = (mensaje or "").strip()
    url_app = f"whatsapp://send?phone={numero}"
    if texto:
        url_app += f"&text={quote(texto)}"

    url_web = f"https://wa.me/{numero}"
    if texto:
        url_web += f"?text={quote(texto)}"

    def cerrar_dialogo(e):
        dialogo.open = False
        page.update()

    async def lanzar_url(url):
        resultado = page.launch_url(url)
        if inspect.isawaitable(resultado):
            await resultado

    async def continuar_a_whatsapp(e):
        dialogo.open = False
        page.update()
        try:
            plataforma = getattr(page, "platform", None)
            es_android = plataforma == ft.PagePlatform.ANDROID or "android" in str(plataforma).lower()

            if es_android:
                await lanzar_url(url_app)
                return

            if webbrowser.open(url_web, new=2):
                return
            await lanzar_url(url_web)
        except Exception:
            try:
                await lanzar_url(url_web)
            except Exception:
                pass

    async def abrir_enlace_web(e):
        dialogo.open = False
        page.update()
        try:
            await lanzar_url(url_web)
        except Exception:
            pass

    dialogo = ft.AlertDialog(
        modal=True,
        title=ft.Text("Contactar con la tienda 💬", size=20, weight=ft.FontWeight.BOLD),
        content=ft.Text(
            "¿Quieres escribirle a la tienda por WhatsApp? 📲\n\n"
            "Te ayudaremos a resolver tus dudas sobre este producto 😊"
        ),
        actions=[
            ft.TextButton("Ahora no", on_click=cerrar_dialogo),
            ft.TextButton("Abrir en navegador 🌐", on_click=abrir_enlace_web),
            ft.ElevatedButton("Escribir por WhatsApp 💚", on_click=continuar_a_whatsapp),
        ],
        actions_alignment=ft.MainAxisAlignment.END,
    )
    page.overlay.append(dialogo)
    dialogo.open = True
    page.update()
    return True


async def abrir_url(page: ft.Page, url: str | None):
    if not url:
        return False
    try:
        plataforma = str(getattr(page, "platform", "")).lower()
        es_movil = "android" in plataforma or "ios" in plataforma

        if not es_movil and webbrowser.open(url, new=2):
            return True

        launcher = getattr(page, "launch_url", None)
        if launcher is None:
            return False
        resultado = launcher(url)
        if inspect.isawaitable(resultado):
            await resultado
        return True
    except Exception as exc:
        print(f"❌ No se pudo abrir la URL {url}: {exc}")
        return False


def abrir_url_en_background(page: ft.Page, url: str | None):
    if not url:
        return False

    def ejecutar_apertura():
        try:
            asyncio.run(abrir_url(page, url))
        except Exception as exc:
            print(f"❌ No se pudo abrir la URL en segundo plano {url}: {exc}")

    threading.Thread(target=ejecutar_apertura, daemon=True).start()
    return True


def _version_comparable(version):
    try:
        return tuple(int(parte) for parte in str(version).strip().lstrip("v").split("."))
    except (TypeError, ValueError):
        return ()


async def comprobar_actualizacion(page: ft.Page):
    """Muestra una actualización disponible en cada arranque, sin guardar estado."""
    try:
        await asyncio.sleep(1.5)
        from supabase_config import APP_VERSION as CONFIGURED_APP_VERSION, SUPABASE_UPDATE_URL

        update_url = SUPABASE_UPDATE_URL
        if not update_url:
            return

        def descargar_manifest():
            with urlopen(update_url, timeout=6) as response:
                return json.loads(response.read().decode("utf-8"))

        manifest = await asyncio.to_thread(descargar_manifest)
        version_remota = manifest.get("version")
        enlace = manifest.get("url")
        version_local = CONFIGURED_APP_VERSION or APP_VERSION
        if not enlace or _version_comparable(version_remota) <= _version_comparable(version_local):
            return

        def cerrar(e):
            aviso.open = False
            page.update()

        async def actualizar(e):
            aviso.open = False
            page.update()
            await abrir_url(page, enlace)

        aviso = ft.AlertDialog(
            modal=False,
            title=ft.Text("Nueva versión disponible", weight=ft.FontWeight.BOLD, color="#040D4A"),
            content=ft.Text(
                f"SmartStore tiene disponible la versión {version_remota}.\n"
                "Puedes actualizarla cuando quieras."
            ),
            actions=[
                ft.TextButton("Cerrar", on_click=cerrar),
                ft.ElevatedButton("Actualizar", on_click=actualizar, bgcolor="#168AAD", color="white"),
            ],
            actions_alignment=ft.MainAxisAlignment.END,
        )
        page.overlay.append(aviso)
        aviso.open = True
        page.update()
    except Exception as exc:
        log_error(f"No se pudo comprobar actualizacion: {exc}", exc_info=False)
        return False

            except Exception:
                pass
            try:
                if dialogo in list(getattr(page, "overlay", [])):
                    page.overlay.remove(dialogo)
            except Exception:
                pass
    except Exception:
        pass

    try:
        if hasattr(page, "overlay") and page.overlay:
            for item in list(page.overlay):
                try:
                    if hasattr(item, "open") and item is not dialogo:
                        item.open = False
                except Exception:
                    pass
    except Exception:
        pass

    try:
        if hasattr(page, "update"):
            page.update()
    except Exception:
        pass

    # Diálogo de escritorio: botones, colores y acciones para cámara o galería.

                    page.services.remove(file_picker)

            page.update()

    dialogo = ft.AlertDialog(
        modal=True,
        title=ft.Text(title),

            if not archivos:
                finalizar_cancelacion()
                return

            archivo = archivos[0]
            datos = archivo.bytes
            ruta = archivo.path
            extension = os.path.splitext(archivo.name or ruta or "")[1].lower() or ".jpg"
            if datos:
                temp_dir = _obtener_directorio_imagenes() / "temp"
                temp_dir.mkdir(parents=True, exist_ok=True)
                ruta = temp_dir / f"seleccion_{datetime.now().strftime('%Y%m%d_%H%M%S_%f')}{extension}"
                with open(ruta, "wb") as image_file:
                    image_file.write(bytes(datos))

            if ruta:
                callback(str(Path(ruta).resolve()))
        except Exception as exc:
            mostrar_snackbar(page, f"❌ Error al abrir la galería: {exc}", color="red")
        finally:
            try:
                if file_picker in page.services:
                    page.services.remove(file_picker)
            except Exception:
                pass

    def abrir_camara(e):
        cerrar_dialogo()
        page.update()
        if not _es_android(page):
            _mostrar_aviso_camara_no_disponible(page, on_cancel=on_cancel)
            return
        if not CAMERA_DISPONIBLE:
            _mostrar_aviso_camara_no_disponible(page, on_cancel=on_cancel)
            return
        _abrir_camara_android(page, callback, on_cancel=finalizar_cancelacion)

    acciones = [
        ft.ElevatedButton("📷 Cámara", on_click=abrir_camara, bgcolor="#040D4A", color="white"),
        ft.ElevatedButton("🖼️ Galería", on_click=abrir_galeria, bgcolor="#168AAD", color="white"),

    if dialogo not in list(getattr(page, "overlay", [])):
        page.overlay.append(dialogo)
    dialogo.open = True
    page.update()


def _abrir_selector_windows(page: ft.Page, callback, title="Seleccionar imagen", allowed_extensions=None, on_cancel=None):
    """Selector de archivos para Windows usando ctypes"""
    try:
        import ctypes
        from ctypes import wintypes

        class OPENFILENAMEW(ctypes.Structure):
            _fields_ = [
                ("lStructSize", wintypes.DWORD),
                ("hwndOwner", wintypes.HWND),
                ("hInstance", wintypes.HINSTANCE),
                ("lpstrFilter", wintypes.LPCWSTR),
                ("lpstrCustomFilter", wintypes.LPWSTR),
                ("nMaxCustFilter", wintypes.DWORD),
                ("nFilterIndex", wintypes.DWORD),
                ("lpstrFile", wintypes.LPWSTR),
                ("nMaxFile", wintypes.DWORD),
                ("lpstrFileTitle", wintypes.LPWSTR),
                ("nMaxFileTitle", wintypes.DWORD),
                ("lpstrInitialDir", wintypes.LPCWSTR),
                ("lpstrTitle", wintypes.LPCWSTR),
                ("Flags", wintypes.DWORD),
                ("nFileOffset", wintypes.WORD),
                ("nFileExtension", wintypes.WORD),
                ("lpstrDefExt", wintypes.LPCWSTR),
                ("lCustData", wintypes.LPARAM),
                ("lpfnHook", wintypes.LPVOID),
                ("lpTemplateName", wintypes.LPCWSTR),
                ("pvReserved", wintypes.LPVOID),
                ("dwReserved", wintypes.DWORD),
                ("FlagsEx", wintypes.DWORD),
            ]

        filter_string = "\0".join([f"Imagen\0*.jpg;*.jpeg;*.png;*.webp"]) + "\0\0"
        buffer = ctypes.create_unicode_buffer(260)
        ofn = OPENFILENAMEW()
        ofn.lStructSize = ctypes.sizeof(ofn)
        ofn.hwndOwner = None
        ofn.lpstrFilter = filter_string
        ofn.lpstrFile = ctypes.cast(buffer, wintypes.LPWSTR)
        ofn.nMaxFile = len(buffer)
        ofn.lpstrFileTitle = None
        ofn.nMaxFileTitle = 0
        ofn.lpstrInitialDir = None
        ofn.lpstrTitle = title
        ofn.Flags = 0x00001000 | 0x00000800 | 0x00000008
        ofn.lpstrDefExt = None
        ofn.nFilterIndex = 1
        ofn.lCustData = 0
        ofn.lpfnHook = None
        ofn.lpTemplateName = None
        ofn.pvReserved = None
        ofn.dwReserved = 0
        ofn.FlagsEx = 0

        if ctypes.windll.comdlg32.GetOpenFileNameW(ctypes.byref(ofn)):
            callback(buffer.value)
        else:
            mostrar_snackbar(page, "Selección cancelada. Puedes intentarlo nuevamente 😊", color="orange")
            if on_cancel:
                on_cancel()
    except Exception as exc:
        print(f"❌ Error en selector Windows: {exc}")
        mostrar_snackbar(page, f"Error: {exc}", color="red")


def _abrir_camara_android(page: ft.Page, callback, on_cancel=None):
    """
    Abre la cámara en Android usando el mismo flujo que ya funciona
    en registrar producto y crear tienda.
    """
    prev_controls = list(page.controls)
    file_picker = ft.FilePicker()
    page.services.append(file_picker)

    camera_container = ft.Container(
        width=380,
        height=400,
        bgcolor="#1a1a2e",
        border_radius=10,
        visible=False,
    )
    camera = None

    def restaurar_pagina():
        page.controls.clear()
        page.controls.extend(prev_controls)
        page.update()

    def cerrar_con_retorno():
        try:
            page.controls.clear()
            page.controls.extend(prev_controls)
        except Exception:
            pass
        page.update()
        if on_cancel:
            on_cancel()

    def normalizar_datos_imagen(image_data):
        if image_data is None:
            return None
        if isinstance(image_data, str):
            try:
                return base64.b64decode(image_data.split(",", 1)[-1], validate=True)
            except (ValueError, base64.binascii.Error):
                return None
        try:
            datos = bytes(image_data)
        except (TypeError, ValueError):
            return None
        return datos or None

    def guardar_imagen_capturada(image_data):
        datos = normalizar_datos_imagen(image_data)
        if not datos:
            return None
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S_%f")
        temp_dir = _obtener_directorio_imagenes() / "temp"
        temp_dir.mkdir(parents=True, exist_ok=True)
        ruta = temp_dir / f"foto_{timestamp}.jpg"
        with open(ruta, "wb") as image_file:
            image_file.write(datos)
        return str(ruta.resolve())

    async def limpiar_camara():
        nonlocal camera
        if camera:
            try:
                await camera.pause_preview()
            except Exception:
                pass
        camera_container.content = None
        camera_container.visible = False
        camera = None

    async def abrir_galeria_accion(e=None):
        try:
            archivos = await file_picker.pick_files(
                allow_multiple=False,
                file_type=ft.FilePickerFileType.IMAGE,
                with_data=True,
            )
            if not archivos:
                mostrar_snackbar(page, "Selección cancelada", color="orange")
                return

            archivo = archivos[0]
            datos_imagen = normalizar_datos_imagen(archivo.bytes)
            extension = os.path.splitext(archivo.name or archivo.path or "")[1].lower() or ".jpg"
            ruta = archivo.path

            if datos_imagen:
                temp_dir = _obtener_directorio_imagenes() / "temp"
                temp_dir.mkdir(parents=True, exist_ok=True)
                ruta = temp_dir / f"seleccion_{datetime.now().strftime('%Y%m%d_%H%M%S_%f')}{extension}"
                with open(ruta, "wb") as image_file:
                    image_file.write(datos_imagen)

            if not ruta:
                mostrar_snackbar(page, "❌ Android no entregó la imagen seleccionada", color="red")
                return

            await limpiar_camara()
            restaurar_pagina()
            callback(str(Path(ruta).resolve()))
        except Exception as ex:
            mostrar_snackbar(page, f"❌ Error al seleccionar imagen: {ex}", color="red")

    async def inicializar_camara():
        nonlocal camera
        try:
            camera = fc.Camera(expand=True, preview_enabled=True)
            camera_container.content = camera
            camera_container.visible = True
            page.update()

            cameras = await camera.get_available_cameras()
            if not cameras:
                raise RuntimeError("No se encontró ninguna cámara")
            await camera.initialize(cameras[0], fc.ResolutionPreset.MEDIUM, enable_audio=False)
        except Exception as ex:
            camera_container.visible = False
            mostrar_snackbar(page, f"❌ Error al inicializar cámara: {ex}", color="red")
            page.update()
            cerrar_con_retorno()

    async def _do_capture():
        nonlocal camera
        if not camera:
            return
        try:
            mostrar_snackbar(page, "Capturando tu imagen... 📸", color="#040D4A")
            page.update()
            image_data = await camera.take_picture()
            ruta = guardar_imagen_capturada(image_data)
            await limpiar_camara()
            restaurar_pagina()
            if ruta:
                callback(str(Path(ruta).resolve()))
            if not ruta:
                mostrar_snackbar(page, "No se capturó ninguna imagen. Inténtalo nuevamente 😊", color="orange")
        except Exception as ex:
            print(f"❌ Error al capturar imagen async: {ex}")
            mostrar_snackbar(page, f"Error al capturar imagen: {ex}", color="red")
            page.update()
            cerrar_con_retorno()

    def capturar(e):
        try:
            asyncio.create_task(_do_capture())
        except Exception as ex:
            print(f"❌ Error al iniciar captura: {ex}")
            mostrar_snackbar(page, f"Error al iniciar captura: {ex}", color="red")
            page.update()

    def cerrar_camara(e=None):
        try:
            asyncio.create_task(limpiar_camara())
        except Exception:
            pass
        restaurar_pagina()

    page.clean()
    page.add(
        ft.SafeArea(
            ft.Column(
                [
                    ft.Text("📸 Seleccionar imagen", size=24, weight=ft.FontWeight.BOLD),
                    camera_container,
                    ft.Divider(height=20, color="transparent"),
                    ft.Row(
                        [
                            ft.ElevatedButton(
                                "🖼️ Abrir Galería",
                                on_click=abrir_galeria_accion,
                                bgcolor="#168AAD",
                                color="white",
                                width=180,
                            )
                        ],
                        alignment=ft.MainAxisAlignment.CENTER,
                        spacing=10,
                    ),
                    ft.Row(
                        [
                            ft.ElevatedButton(
                                "📸 Capturar",
                                on_click=capturar,
                                bgcolor="#EFA520",
                                color="white",
                                width=180,
                            ),
                            ft.ElevatedButton(
                                "↩️ Cerrar",
                                on_click=cerrar_camara,
                                bgcolor="red",
                                color="white",
                                width=180,
                            ),
                        ],
                        alignment=ft.MainAxisAlignment.CENTER,
                        spacing=10,
                    ),
                ],
                horizontal_alignment=ft.CrossAxisAlignment.CENTER,
                alignment=ft.MainAxisAlignment.CENTER,
                spacing=10,
                expand=True,
            )
        )
    )
    page.update()
    asyncio.create_task(inicializar_camara())

from flask import Flask, request, jsonify
import database as db

app = Flask(__name__)

@app.route('/webhook', methods=['POST'])
def webhook():
    # Recibe eventos de PayPal y actualiza el estado del pedido correspondiente.
    try:
        data = request.get_json()
        event_type = data.get('event_type', '')

        if event_type == 'PAYMENT.CAPTURE.COMPLETED':
            resource = data.get('resource', {})

            # Intentar obtener el order_id desde supplementary_data (ruta correcta de PayPal)
            order_id = None
            supplementary = resource.get('supplementary_data', {})
            related_ids = supplementary.get('related_ids', {})
            order_id = related_ids.get('order_id')

            # Fallback: intentar desde invoice_id (por compatibilidad)
            if not order_id:
                order_id = resource.get('invoice_id')

            if order_id:
                db.actualizar_estado_pago(order_id)
                print(f"✅ Webhook: Pago procesado para orden {order_id}")
                return jsonify({'status': 'ok'}), 200
            else:
                print("⚠️ Webhook: No se encontró order_id en el evento")
                return jsonify({'status': 'missing_order_id'}), 400

        return jsonify({'status': 'ignored'}), 200

    except Exception as e:
        print(f"❌ Webhook error: {str(e)}")
        return jsonify({'status': 'error', 'message': str(e)}), 500

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)
